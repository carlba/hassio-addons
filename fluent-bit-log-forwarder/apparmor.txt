#include <tunables/global>

# NOTE: profiles were validated in `complain` mode (all four log sources,
# TLS on/off) before enforcement was enabled below. An earlier enforcement
# attempt on 2026-09-10 failed with `/bin/sh: 0: cannot open /init:
# Permission denied` — root-caused via `journalctl -g 'apparmor='` to a
# `requested_mask="r"` denial on `/init` itself: it's a `#!/bin/sh` script,
# and the interpreter needs to open+read it, which plain `ix` doesn't grant.
# Fixed by using `rix` on that rule.

profile fluent_bit_log_forwarder flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/python>
  # Covers /etc/resolv.conf, nsswitch.conf, hosts, host.conf, gai.conf, NSS
  # libs, and DNS network access in one go — used by curl (Supervisor API)
  # and bash's own name lookups. Standard abstraction instead of hand-
  # listing each glibc resolver config file one denial at a time.
  #include <abstractions/nameservice>
  # Covers bash's own startup file reads (e.g. /etc/bash.bashrc)
  #include <abstractions/bash>
  # curl's TLS stack (openssl.cnf, engines) and CA cert bundle reads —
  # replaces hand-listing /etc/ssl/**, ca-certificates paths one at a time
  #include <abstractions/openssl>
  #include <abstractions/ssl_certs>
  # /dev/tty and friends
  #include <abstractions/consoles>

  # Capabilities
  # NOTE: no bare `file,` rule here (unlike a stock HA app template) — it
  # grants blanket rwlkmix access to every path, which apparmor_parser
  # can't merge with the `cx ->` transition rule below ("merged rule with
  # conflicting x modifiers", fails to load). The specific path rules
  # throughout this profile already cover what the app needs.
  signal (send) set=(kill,term,int,hup,cont),

  # bashio::config / bashio::api.supervisor calls curl against the
  # Supervisor API (hassio_api: true in config.yaml) to read this addon's
  # own options — needs network access same as the fluent_bit child
  # profile's own HTTP calls to Loki below. DNS/TLS/CA certs covered by
  # the abstractions above.
  network inet stream,
  network inet6 stream,

  # S6-Overlay
  # `/init` is a `#!/bin/sh` script (see rootfs of base-debian:trixie), so
  # the interpreter needs to open+read it as a file in addition to the
  # exec bit — plain `ix` alone denies that open with requested_mask="r".
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  # pip-installed jinja2-cli entry point (#!/usr/bin/python3 shebang, same
  # open+read requirement as /init) plus the packages it imports; the bare
  # dir entry needs `r` too (python lists it), same pattern as services.d
  /usr/local/bin/ r,
  /usr/local/bin/** rix,
  /usr/local/lib/python3*/dist-packages/** r,
  /run/{s6,s6-rc*,service}/** ix,
  # rix, not ix: several s6-overlay scripts under /package (e.g. preinit,
  # stage0/1/2) are #!/bin/sh scripts, and the interpreter needs to open+
  # read them, which plain `ix` doesn't grant (same issue as /init above).
  /package/** rix,
  /command/** ix,
  # s6-ls lists these directories at startup; `**` alone doesn't cover the
  # directory entry itself (name="/etc/services.d/", trailing slash)
  /etc/fix-attrs.d/ r,
  /etc/fix-attrs.d/** rwix,
  /etc/services.d/ r,
  /etc/services.d/** rwix,
  /etc/cont-init.d/ r,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/ r,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,

  # s6-overlay-suexec reads these to resolve the container's user/group
  /etc/passwd r,
  /etc/group r,

  # s6-rc-compile reads the service definitions to build the runtime
  # service database
  /etc/s6-overlay/ r,
  /etc/s6-overlay/** r,

  # Bashio
  # /usr/bin/bashio -> /usr/lib/bashio/bashio, itself a #!/usr/bin/env bash
  # script — same open+read requirement as /init, so rix not ix.
  /usr/lib/bashio/** rix,
  /tmp/** rwk,

  # run script renders the Fluent Bit config here via `mkdir -p` + jinja2
  /etc/fluent-bit/ rw,
  /etc/fluent-bit/** rw,

  # Access to options.json and other files within your app
  /data/** rw,

  # Home Assistant Core config mount (read-only)
  /homeassistant/** r,

  # jinja2-cli (run.py, via `exec python3` in its s6 run script) renders
  # the Fluent Bit config template. Runs directly in this profile rather
  # than a `cx`-transitioned child — it's our own script, not a separate
  # trust boundary like fluent-bit, and a python3 child profile would need
  # its own transition rule that overlaps /usr/bin/** ix above the same way
  # a bare `file,` rule does.
  /usr/lib/python3*/** r,
  /usr/lib/python3.*/lib-dynload/*.so mr,

  # Locate the active journal directory under /var/log/journal to pass to
  # Fluent Bit's `systemd` input
  /var/log/journal/ r,
  /var/log/journal/** r,
  /proc/sys/kernel/random/uuid r,
  /proc/*/maps r,
  /proc/*/status r,
  /dev/urandom r,
  /dev/random r,

  # Start new profile for the fluent-bit service
  /usr/bin/fluent-bit cx -> fluent_bit,

  profile fluent_bit flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/openssl>
    #include <abstractions/nameservice>
    #include <abstractions/ssl_certs>
    #include <abstractions/consoles>

    # Receive signals from S6-Overlay
    signal (receive) peer=*_fluent_bit_log_forwarder,

    network inet stream,
    network inet6 stream,

    # Access to options.json and other files within your app; `k` (lock) is
    # required for the systemd input's sqlite cursor DB at /data/journal.db.
    # Bare dir entry (name="/data/") also needed: fluent-bit opens the
    # directory itself, which `**` alone doesn't cover (same pattern as
    # /etc/services.d/ etc. in the parent profile above).
    /data/ r,
    /data/** rwk,

    # Fluent Bit config, parsers, and helper scripts
    /etc/fluent-bit/** rix,

    # System journal, read directly via the `systemd` input; the DB file
    # under /data tracks the read cursor across restarts
    /var/log/journal/ r,
    /var/log/journal/** r,

    # Access required for service functionality
    # `m` (mmap-exec): fluent-bit mmaps its own binary at startup
    /usr/bin/fluent-bit mr,
    /etc/passwd r,
    /proc/sys/kernel/random/uuid r,
    /proc/cpuinfo r,
    /proc/*/maps r,
    /proc/*/status r,
    /sys/devices/system/cpu/** r,
    /dev/urandom r,
    /dev/random r,
  }
}
