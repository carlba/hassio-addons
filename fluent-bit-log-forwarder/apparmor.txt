#include <tunables/global>

# NOTE: profiles below are flagged `complain` for initial validation.
# 1. Deploy, exercise the app normally (all four log sources, TLS on/off).
# 2. Review the audit log: journalctl _TRANSPORT="audit" -g 'apparmor="ALLOWED"'
# 3. Add any missing access below, then remove `,complain` from each profile
#    flags line before relying on this for confinement.

profile fluent_bit_log_forwarder flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),
  network inet stream,
  network inet6 stream,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within your app
  /data/** rw,

  # Home Assistant Core config mount (read-only)
  /homeassistant/** r,

  # Log files written by supervisor-poller (python3 profile below) and
  # tailed by fluent-bit (fluent_bit profile below)
  /var/log/fluent-bit-src/ rw,
  /var/log/fluent-bit-src/** rw,

  # Start new profile for the fluent-bit service
  /usr/bin/fluent-bit cx -> fluent_bit,

  # Start new profile for the supervisor-poller service
  /usr/bin/python3.[0-9]* cx -> python3,
  /usr/bin/python3 cx -> python3,

  profile fluent_bit flags=(attach_disconnected,mediate_deleted,complain) {
    #include <abstractions/base>
    #include <abstractions/openssl>

    # Receive signals from S6-Overlay
    signal (receive) peer=*_fluent_bit_log_forwarder,

    network inet stream,
    network inet6 stream,

    # Access to options.json and other files within your app
    /data/** rw,

    # Fluent Bit config, parsers, and helper scripts
    /etc/fluent-bit/** rix,

    # Log source files written by supervisor-poller, tailed via the `tail`
    # input; DB files track tail offsets across restarts
    /var/log/fluent-bit-src/*.log r,

    # Access required for service functionality
    /usr/bin/fluent-bit r,
    /etc/passwd r,
    /etc/resolv.conf r,
    /etc/ssl/** r,
    /etc/ssl/openssl.cnf r,
    /usr/lib/ssl/** r,
    /usr/share/ca-certificates/** r,
    /etc/ca-certificates.conf r,
    /proc/sys/kernel/random/uuid r,
    /proc/cpuinfo r,
    /proc/*/maps r,
    /proc/*/status r,
    /sys/devices/system/cpu/** r,
    /dev/urandom r,
    /dev/random r,
    /dev/tty rw,
  }

  profile python3 flags=(attach_disconnected,mediate_deleted,complain) {
    #include <abstractions/base>
    #include <abstractions/python>
    #include <abstractions/openssl>

    # Receive signals from S6-Overlay
    signal (receive) peer=*_fluent_bit_log_forwarder,

    network inet stream,
    network inet6 stream,

    # run.py itself, plus the stdlib it imports (json, threading,
    # urllib.request/error, os, sys, time)
    /etc/services.d/supervisor-poller/run.py r,
    /usr/bin/python3.[0-9]* rix,
    /usr/lib/python3*/** r,
    /usr/lib/python3.*/lib-dynload/*.so mr,

    # Writes the log files fluent-bit tails; creates the dir and files
    # (os.makedirs, open(..., "a")/"ab") on each stream (re)start
    /var/log/fluent-bit-src/ rw,
    /var/log/fluent-bit-src/*.log rw,

    # HTTP(S) calls to the Supervisor API (http://supervisor, no TLS) via
    # urllib — needs cert store even though the connection itself is plain
    # HTTP, since Python's ssl module loads it at import time
    /etc/passwd r,
    /etc/resolv.conf r,
    /etc/ssl/** r,
    /etc/ssl/openssl.cnf r,
    /usr/lib/ssl/** r,
    /usr/share/ca-certificates/** r,
    /etc/ca-certificates.conf r,
    /proc/sys/kernel/random/uuid r,
    /proc/*/maps r,
    /proc/*/status r,
    /dev/urandom r,
    /dev/random r,
  }
}
